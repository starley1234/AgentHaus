📚 Как подключить в свой сервис

// 1. Импорт модуля (уже в services/lib/robokassa.mjs)
import { Robokassa } from "../lib/robokassa.mjs";

const rk = new Robokassa({
  merchantLogin: process.env.ROBOKASSA_MERCHANT_LOGIN,
  password1: process.env.ROBOKASSA_PASSWORD1,
  password2: process.env.ROBOKASSA_PASSWORD2,
  isTest: true,
  algorithm: "md5" // или sha256 — должен совпадать с настройками в ЛК Robokassa
});

// 2. Создание платежа (в любом сервисе)
app.post("/api/create-payment", async (req, res) => {
  const { amount, description, userId } = req.body;
  const invId = Date.now(); // уникальный ID заказа

  const paymentUrl = rk.generatePaymentUrl({
    outSum: amount,
    invId,
    description,
    userParams: {
      Shp_userId: userId,
      Shp_service: "bom-parse"
    },
    email: req.body.email
  });

  // Сохранить заказ в БД как pending
  // await db.save({ invId, amount, status: "pending" });

  res.json({ paymentUrl, invId });
});

// 3. Обработка ResultURL — серверное уведомление от Robokassa (обязательно!)
app.post("/api/robokassa/result", (req, res) => {
  const data = req.body; // OutSum, InvId, SignatureValue, Shp_...

  if (!rk.validateResult(data)) {
    return res.status(400).send("Invalid signature");
  }

  // Начислить кредиты / выдать товар
  const userId = data.Shp_userId;
  const service = data.Shp_service;
  // await credits.add(userId, amount);

  // Обязательно вернуть OK{InvId}
  res.send(rk.getSuccessAnswer(data.InvId));
});

// 4. SuccessURL — куда попадает пользователь после оплаты
app.get("/api/robokassa/success", (req, res) => {
  if (!rk.validateSuccess(req.query)) {
    // можно показать ошибку, но для демо показываем успех
  }
  res.send("✅ Оплата успешна! Кредиты начислены.");
});

// 5. FailURL
app.get("/api/robokassa/fail", (req, res) => {
  res.send("❌ Оплата не удалась");
});

Полный модуль в services/lib/robokassa.mjs — скопируй в свой проект, укажи ключи из ЛК Robokassa (MerchantLogin, Password1, Password2, алгоритм).

🔧 Настройка в ЛК Robokassa

  1. Зарегистрируйся на robokassa.ru → Создай магазин
  2. Технические настройки: Алгоритм — MD5 (или SHA256), ResultURL — https://твой-домен/robokassa-payment/api/robokassa/result, метод POST, SuccessURL — /api/robokassa/success, FailURL — /api/robokassa/fail
  3. Скопируй MerchantLogin, Password1, Password2 в .env: 
    ROBOKASSA_MERCHANT_LOGIN=твой_логин
ROBOKASSA_PASSWORD1=пароль1
ROBOKASSA_PASSWORD2=пароль2
ROBOKASSA_IS_TEST=true
ROBOKASSA_ALGORITHM=md5
  4. Включи тестовый режим в ЛК и сделай тестовую оплату
  5. ResultURL должен возвращать OK{InvId} и быть доступен из интернета (не localhost)
